cmake_minimum_required(VERSION 3.22)
find_package(Git REQUIRED)
file(READ "${CMAKE_CURRENT_LIST_DIR}/sdk.lock.json" SDK_LOCK)
string(JSON EXPECTED_IDF_COMMIT GET "${SDK_LOCK}" commit)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "$ENV{IDF_PATH}" rev-parse HEAD
    OUTPUT_VARIABLE ACTUAL_IDF_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE SDK_RESULT)
if(NOT SDK_RESULT EQUAL 0 OR NOT ACTUAL_IDF_COMMIT STREQUAL EXPECTED_IDF_COMMIT)
    message(FATAL_ERROR "ESP-IDF commit differs from sdk.lock.json")
endif()
set(IDF_TARGET esp32 CACHE STRING "Only the verified ESP32 target is supported")
set(COMPONENTS main)
set(PROJECT_VER "0.1.0")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(gamexr_usb_diagnostics)
if(NOT IDF_VERSION_MAJOR EQUAL 6 OR NOT IDF_VERSION_MINOR EQUAL 1)
    message(FATAL_ERROR "This project requires the pinned ESP-IDF 6.1 SDK")
endif()
if(NOT IDF_TARGET STREQUAL "esp32")
    message(FATAL_ERROR "Only esp32 is supported by this board contract")
endif()
