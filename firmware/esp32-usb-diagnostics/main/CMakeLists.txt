idf_component_register(SRCS "app_main.c" "sensors.c" "mpu6500.c" "telemetry.c" "network.c" "imu_window.c" "device_auth.c" "ota_policy.c" "ota_guard.c" "ota_update.c" "bench.c" "bench_uart.c" "wifi_bench.c" "${CMAKE_CURRENT_LIST_DIR}/../generated/cockpit_assets.c"
    INCLUDE_DIRS "." "../generated"
    REQUIRES app_update mbedtls spi_flash bootloader_support esp_driver_gpio esp_driver_spi esp_driver_uart esp_adc esp_timer esp_app_format esp_wifi esp_netif esp_event esp_http_server esp_https_server)
target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Werror)
target_add_binary_data(${COMPONENT_LIB} "../web/index.html" BINARY RENAME_TO "device_page")
target_add_binary_data(${COMPONENT_LIB} "../web/device.js" BINARY RENAME_TO "device_script")
set(PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../private")
if(EXISTS "${PRIVATE}/server.pem")
    foreach(NAME server.pem server-key.pem ca.cer control-key.txt)
        if(NOT EXISTS "${PRIVATE}/${NAME}")
            message(FATAL_ERROR "TLS files incomplete; use the documented private build inputs")
        endif()
    endforeach()
    target_add_binary_data(${COMPONENT_LIB} "${PRIVATE}/server.pem" TEXT RENAME_TO "device_cert")
    target_add_binary_data(${COMPONENT_LIB} "${PRIVATE}/server-key.pem" TEXT RENAME_TO "device_key")
    target_add_binary_data(${COMPONENT_LIB} "${PRIVATE}/ca.cer" BINARY RENAME_TO "device_ca")
    target_add_binary_data(${COMPONENT_LIB} "${PRIVATE}/control-key.txt" BINARY RENAME_TO "control_key")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE DEVICE_TLS_AVAILABLE=1)
else()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE DEVICE_TLS_AVAILABLE=0)
endif()

include("${CMAKE_CURRENT_LIST_DIR}/../generated/assets.cmake")

# Digest input is generated from the pinned rollback-enabled bootloader, never guessed.
foreach(NAME ota-bootloader.sha256 ota-partitions.sha256)
    if(EXISTS "${PRIVATE}/${NAME}")
        file(READ "${PRIVATE}/${NAME}" DIGEST)
        string(STRIP "${DIGEST}" DIGEST)
        string(LENGTH "${DIGEST}" DIGEST_LENGTH)
        if(NOT DIGEST_LENGTH EQUAL 64 OR NOT DIGEST MATCHES "^[0-9a-f]+$")
            message(FATAL_ERROR "Invalid OTA bootstrap digest")
        endif()
    else()
        set(DIGEST "")
    endif()
    if(NAME STREQUAL "ota-bootloader.sha256")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE OTA_BOOTLOADER_SHA256="${DIGEST}")
    else()
        target_compile_definitions(${COMPONENT_LIB} PRIVATE OTA_PARTITIONS_SHA256="${DIGEST}")
    endif()
endforeach()
